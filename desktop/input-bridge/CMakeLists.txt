cmake_minimum_required(VERSION 3.20)
project(SynergyInputBridge VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED)
add_executable(synergy-input-bridge main.cpp)
target_include_directories(synergy-input-bridge PRIVATE ../../native/include)
target_link_libraries(synergy-input-bridge PRIVATE OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
  target_compile_definitions(synergy-input-bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
  target_link_libraries(synergy-input-bridge PRIVATE ws2_32 xinput9_1_0)
endif()
if(MSVC)
  target_compile_options(synergy-input-bridge PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(synergy-input-bridge PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
