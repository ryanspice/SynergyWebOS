cmake_minimum_required(VERSION 3.20)
project(SynergyWebOSNative VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SYNERGY_WEBOS "Build for a real ARM32 webOS SDK" OFF)
option(SYNERGY_BUILD_UI "Build SDL2/raw GLES2 management app" ON)
option(SYNERGY_STRICT "Warnings are errors" ON)
option(SYNERGY_SANITIZE "Address and undefined behavior sanitizers" OFF)
if(NOT UNIX)
  message(FATAL_ERROR "Use WSL2/Linux: the client and its Unix-domain broker target Linux/webOS.")
endif()
if(SYNERGY_WEBOS AND (NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^[aA][rR][mM]" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4))
  message(FATAL_ERROR "SYNERGY_WEBOS needs a real ARM32 cross-toolchain; host output is not a TV build.")
endif()
find_package(Threads REQUIRED)
find_path(JSONC_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSONC_LIBRARY NAMES json-c REQUIRED)
function(strict target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
  if(SYNERGY_STRICT)
    target_compile_options(${target} PRIVATE -Werror)
  endif()
  if(SYNERGY_SANITIZE)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()
add_library(synergy_core src/config.cpp src/protocol.cpp src/input.cpp src/platform.cpp src/client.cpp src/ipc.cpp foundation/src/geometry.cpp)
target_include_directories(synergy_core PUBLIC include foundation/include ${JSONC_INCLUDE_DIR})
target_link_libraries(synergy_core PUBLIC ${JSONC_LIBRARY} Threads::Threads)
target_compile_definitions(synergy_core PUBLIC SYNERGY_VERSION="${PROJECT_VERSION}")
if(SYNERGY_WEBOS)
  target_compile_definitions(synergy_core PUBLIC SYNERGY_WEBOS=1)
endif()
strict(synergy_core)
add_executable(synergy-daemon src/daemon.cpp)
target_link_libraries(synergy-daemon PRIVATE synergy_core)
strict(synergy-daemon)
add_executable(synergy-tvctl src/control.cpp)
target_link_libraries(synergy-tvctl PRIVATE synergy_core)
strict(synergy-tvctl)
if(SYNERGY_BUILD_UI)
  find_package(PNG REQUIRED)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
  find_path(FREETYPE_INCLUDE_DIR ft2build.h PATH_SUFFIXES freetype2)
  if(SYNERGY_WEBOS AND (NOT SDL2_INCLUDE_DIR OR NOT FREETYPE_INCLUDE_DIR))
    message(FATAL_ERROR "Target SDL-webOS and FreeType headers required; no ABI declaration fallback on TV.")
  endif()
  add_executable(synergy-tv src/ui.cpp foundation/src/runtime.cpp foundation/src/render.cpp)
  target_include_directories(synergy-tv PRIVATE foundation/compat)
  if(SDL2_INCLUDE_DIR AND FREETYPE_INCLUDE_DIR)
    target_include_directories(synergy-tv PRIVATE ${SDL2_INCLUDE_DIR} ${FREETYPE_INCLUDE_DIR})
    target_compile_definitions(synergy-tv PRIVATE SPICE_SYSTEM_HEADERS=1)
  else()
    message(WARNING "Desktop-only bootstrap uses Spice Home's narrow SDL2/FreeType ABI declarations.")
  endif()
  target_link_libraries(synergy-tv PRIVATE synergy_core PNG::PNG ${CMAKE_DL_LIBS})
  strict(synergy-tv)
  install(TARGETS synergy-tv RUNTIME DESTINATION .)
endif()
foreach(t synergy-daemon synergy-tvctl)
  set_target_properties(${t} PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
endforeach()
if(TARGET synergy-tv)
  set_target_properties(synergy-tv PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
endif()
install(TARGETS synergy-daemon synergy-tvctl RUNTIME DESTINATION .)
install(FILES appinfo.json icon.png icon-large.png NOTICE.md DESTINATION .)
install(FILES ../LICENSE DESTINATION .)
install(FILES foundation/LICENSE DESTINATION . RENAME FOUNDATION-LICENSE)
install(PROGRAMS scripts/provision-root.sh DESTINATION scripts)
include(CTest)

  target_link_libraries(synergy-tests PRIVATE synergy_core)
  strict(synergy-tests)
  add_test(NAME native-core COMMAND synergy-tests)

  if(Python3_FOUND AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME native-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:synergy-daemon> $<TARGET_FILE:synergy-tvctl>)
    set_tests_properties(native-integration PROPERTIES TIMEOUT 45)
    add_test(NAME native-bundle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bundle_tests.py)
    add_test(NAME gear-settings-spike COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/../spikes/gear-settings -p test_*.py -v)
  endif()
endif()
