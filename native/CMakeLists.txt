if(SYNERGY_OPTIMIZE_SIZE)
  if(SYNERGY_SANITIZE OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Size optimization requires a GNU/Clang release build without sanitizers.")
  endif()
  include(CheckIPOSupported)
  check_ipo_supported(RESULT SYNERGY_IPO_SUPPORTED OUTPUT SYNERGY_IPO_ERROR)
  if(NOT SYNERGY_IPO_SUPPORTED)
    message(FATAL_ERROR "Selected toolchain cannot perform IPO: ${SYNERGY_IPO_ERROR}")
  endif()
endif()

function(strict target)
  if(SYNERGY_OPTIMIZE_SIZE)
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:-Os;-ffunction-sections;-fdata-sections>")
    target_link_options(${target} PRIVATE "$<$<CONFIG:Release>:-Wl,--gc-sections>")
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  endif()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2)
  if(SYNERGY_STRICT)
    target_compile_options(${target} PRIVATE -Werror)
  endif()
  if(SYNERGY_SANITIZE)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()

if(SYNERGY_ENABLE_LS2)
  if(NOT SYNERGY_WEBOS)
    message(FATAL_ERROR "LS2 transport requires the verified webOS target build.")
  endif()
  set(SYNERGY_LS2_INCLUDE_DIR "" CACHE PATH "Pinned official LS2 public headers")
  set(SYNERGY_GLIB_INCLUDE_DIR "" CACHE PATH "Target GLib public headers")
  set(SYNERGY_GLIB_CONFIG_INCLUDE_DIR "" CACHE PATH "Target glibconfig.h directory")
  if(NOT EXISTS "${SYNERGY_LS2_INCLUDE_DIR}/luna-service2/lunaservice.h" OR NOT EXISTS "${SYNERGY_GLIB_INCLUDE_DIR}/glib.h" OR NOT EXISTS "${SYNERGY_GLIB_CONFIG_INCLUDE_DIR}/glibconfig.h")
    message(FATAL_ERROR "LS2 needs pinned official and target GLib headers; no ABI declaration fallback.")
  endif()
  add_executable(synergy-control-bridge src/ls2_service.cpp)
  target_include_directories(synergy-control-bridge PRIVATE ${SYNERGY_LS2_INCLUDE_DIR} ${SYNERGY_GLIB_INCLUDE_DIR} ${SYNERGY_GLIB_CONFIG_INCLUDE_DIR})
  target_link_libraries(synergy-control-bridge PRIVATE synergy_core ${CMAKE_DL_LIBS})
  set_target_properties(synergy-control-bridge PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
  strict(synergy-control-bridge)
  # Helper is intentionally not installed into an ordinary app stage. Explicit
  # reviewed root provisioning and matching hub roles are separate target gates.
endif()
if(SYNERGY_BUILD_UI)
  find_package(PNG REQUIRED)
  find_path(SDL2_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2)
  find_path(FREETYPE_INCLUDE_DIR ft2build.h PATH_SUFFIXES freetype2)
  if(SYNERGY_WEBOS AND (NOT SDL2_INCLUDE_DIR OR NOT FREETYPE_INCLUDE_DIR))
    message(FATAL_ERROR "Target SDL-webOS and FreeType headers required; no ABI declaration fallback on TV.")
  endif()
  add_executable(synergy-tv src/ui.cpp foundation/src/runtime.cpp foundation/src/render.cpp)
  if(SYNERGY_ENABLE_LS2)
    target_sources(synergy-tv PRIVATE src/ls2_transport.cpp)
    target_include_directories(synergy-tv PRIVATE ${SYNERGY_LS2_INCLUDE_DIR} ${SYNERGY_GLIB_INCLUDE_DIR} ${SYNERGY_GLIB_CONFIG_INCLUDE_DIR})
    target_compile_definitions(synergy-tv PRIVATE SYNERGY_USE_LS2=1)
  endif()
  target_include_directories(synergy-tv PRIVATE foundation/compat)
  if(SDL2_INCLUDE_DIR AND FREETYPE_INCLUDE_DIR)
    target_include_directories(synergy-tv PRIVATE ${SDL2_INCLUDE_DIR} ${FREETYPE_INCLUDE_DIR})
    target_compile_definitions(synergy-tv PRIVATE SPICE_SYSTEM_HEADERS=1)
  else()
    message(WARNING "Desktop-only bootstrap uses Spice Home's narrow SDL2/FreeType ABI declarations.")
  endif()
  target_link_libraries(synergy-tv PRIVATE synergy_core PNG::PNG ${CMAKE_DL_LIBS})
  strict(synergy-tv)
  install(TARGETS synergy-tv RUNTIME DESTINATION .)
endif()
